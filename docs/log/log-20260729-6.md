### 5.2 统一行动方案

#### 高优先级 — 消除高斯消元重复

**步骤 1**: 在 `core::compute::matrix` 中添加泛型求解器
```rust
/// Solve A*x = b for real matrices (delegates to solve_linear)
pub fn solve_real(...) -> ... { solve_linear(a, b) }

/// Solve A*x = b for complex matrices
pub fn solve_complex(a: &[Vec<ComplexScalar>], b: &[ComplexScalar]) -> Result<Vec<ComplexScalar>, SimError> {
    // 将复矩阵展开为实矩阵：2n×2n
    let n = a.len();
    let mut real_a = vec![vec![0.0; 2 * n]; 2 * n];
    for i in 0..n {
        for j in 0..n {
            real_a[i][j] = a[i][j].re;
            real_a[i][n + j] = -a[i][j].im;
            real_a[n + i][j] = a[i][j].im;
            real_a[n + i][n + j] = a[i][j].re;
        }
    }
    let mut real_b = vec![0.0; 2 * n];
    for i in 0..n {
        real_b[i] = b[i].re;
        real_b[n + i] = b[i].im;
    }
    let real_x = solve_linear(&real_a, &real_b)?;
    Ok((0..n).map(|i| ComplexScalar::new(real_x[i], real_x[n + i])).collect())
}
```

pub fn gauss_seidel_coupling(...) {

        let mut max_delta = 0.0;

            let old = fields[i].values.clone();
            compute_fn(&mut fields[i])?;
